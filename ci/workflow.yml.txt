# PR 验证 CI：四套件并行跑，判红条件只有一个——出现 ci/baseline/ 之外的新失败。
# 基线内失败完整呈现在日志里；基线维护纪律见 AGENTS.md「CI 基线纪律」。
# 本 workflow 不设分支保护：红不禁止合并，合并仍由业主决定。
name: CI

on:
  pull_request:
  push:
    branches: [master]

concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: true

permissions:
  contents: read

jobs:
  typecheck:
    runs-on: ubuntu-latest
    timeout-minutes: 20
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version-file: .nvmrc
          cache: npm
          cache-dependency-path: |
            package-lock.json
            app/package-lock.json
      - run: npm ci
      - run: npm ci
        working-directory: app
      - run: node ci/check.mjs typecheck

  test-server:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version-file: .nvmrc
          cache: npm
          cache-dependency-path: |
            package-lock.json
            app/package-lock.json
      - run: npm ci
      - run: npm ci
        working-directory: app
      - run: node ci/check.mjs test-server

  test-app:
    runs-on: ubuntu-latest
    timeout-minutes: 20
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version-file: .nvmrc
          cache: npm
          cache-dependency-path: |
            package-lock.json
            app/package-lock.json
      - run: npm ci
      - run: npm ci
        working-directory: app
      - run: node ci/check.mjs test-app

  verify-all:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version-file: .nvmrc
          cache: npm
          cache-dependency-path: |
            package-lock.json
            app/package-lock.json
      - run: npm ci
      - run: npm ci
        working-directory: app
      - run: node ci/check.mjs verify-all
